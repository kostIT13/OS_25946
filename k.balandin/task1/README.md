make clean
make

# 1. Нет аргументов — программа просто ничего не делает
./options

# 2. Группировка опций
./options -ips

# 3. Порядок справа налево: сначала выполнится -s, потом -p, потом -i
./options -i -p -s

# 4. Повторение опций — выведется дважды (в обратном порядке)
./options -v -v

# 5. Установка переменной среды и проверка
./options -v -VTEST=hello | grep TEST

# 6. Некорректное значение для -U
./options -U-5

# 7. Core file size
./options -C1024
./options -c

# 8. Недопустимая опция
./options -x

# 9. Смешанный порядок
./options -d -p -U100 -i
